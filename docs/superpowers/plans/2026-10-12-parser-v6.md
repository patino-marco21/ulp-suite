```ts
/**
 * The stream parsers decode a file's bytes with Buffer.toString('latin1') so that nothing can throw, and the insert writes the string out as UTF-8.
 * A UTF-8 multi-byte character in the source therefore became two or three characters of mojibake ("café" -> "cafÃ©", stored double-encoded: 0.11% of
 * the stored rows), and a login or url written in a non-Latin script was rejected by the mojibake gate. Every line of the pile that holds a byte of
 * 0x80 or more is valid UTF-8, so the decode can be undone exactly: a line is read as UTF-8 when it is valid UTF-8 and stays Latin-1 text when it is not
 * (a legacy single-byte file). Only the 0.55% of the lines with a high byte pay for it.
 *
 * Used by the stream parsers only. parseLine, parseULPContent and parseBlockContent take strings that may already be real Unicode (lib/legacy-repair.ts
 * hands stored rows to parseLine), so they are not changed.
 */

const HIGH_BYTE = /[\u0080-\u00ff]/
const ABOVE_LATIN1 = /[^\u0000-\u00ff]/
const STRICT_UTF8 = new TextDecoder('utf-8', { fatal: true })

export function repairUtf8Line(line: string): string {
  if (!HIGH_BYTE.test(line) || ABOVE_LATIN1.test(line)) return line
  try {
    return STRICT_UTF8.decode(Buffer.from(line, 'latin1'))
  } catch {
    return line
  }
}
```

```ts
 * Line layouts that parseLine used to reject and that parser v6 reads (docs/superpowers/specs/2026-10-12-parser-v6-and-redo-design.md). Each helper is pure and
 * returns null when the line is not of its shape. All of them are tried only AFTER the existing field split found nothing, except splitSpacePaddedColons:
 * the existing split returns white-space-padded fields for that shape, and the garbage gate then rejects them.

/** Ports a web server really listens on: a number after the host that is one of these is a port, not a login (rule D). */
export const COMMON_WEB_PORTS: ReadonlySet<number> = new Set([
  80, 81, 82, 88, 443, 444, 591, 631, 2082, 2083, 2086, 2087, 2095, 2096, 3000, 3001, 3128, 3306, 4443, 5000, 5001, 5432, 5601, 7001, 7443,

/**
 * A "password" that is really a piece of a URL: a scheme, a host name followed by a slash, a leading www., a query string, a web file extension, or three
 * or more slashes. Used by the slash-in-password and protocol-relative rules, where the extra slash can belong to a path instead of the password.
 */
export function looksLikeUrlFragment(password: string): boolean {
  return password.includes('://')
    || /^www\./i.test(password)
    || /^[\w-]+(\.[\w-]+)+\//.test(password)
    || /[?&][\w%.-]+=/.test(password)
    || /\.(php|aspx?|html?|jsp|cgi)(\b|$)/i.test(password)
    || (password.match(/\//g) ?? []).length >= 3
}

const PADDED_COLON = /\s+:\s+/
const EMAIL_SHAPE = /^[^\s/:@]+@[^\s/:@]+\.[^\s/:@]+$/

/**
 * "shop.example.test : alice@mail.test : Secret1": three fields separated by a colon with white space on both sides, no scheme. A first part that is an
 * e-mail address makes it "login : password" (the password may then hold anything). The host needs a dot or a slash and no white space, so a banner such as
 * "Join us : alice : Secret1" or "BTC : <emoji>" is not read. A password that holds another padded colon is ambiguous and refused.
 */
export function splitSpacePaddedColons(line: string): [string, string, string] | null {
  if (line.includes('://')) return null
  const first = PADDED_COLON.exec(line)
  if (!first) return null
  const head = line.slice(0, first.index)
  const afterFirst = line.slice(first.index + first[0].length)
  if (EMAIL_SHAPE.test(head)) {
    if (afterFirst === '' || PADDED_COLON.test(afterFirst)) return null
    return ['', head, afterFirst]
  }
  const second = PADDED_COLON.exec(afterFirst)
  if (!second) return null
  const login = afterFirst.slice(0, second.index)
  const password = afterFirst.slice(second.index + second[0].length)
  if (!/[./]/.test(head) || /\s/.test(head) || /\s/.test(login) || login === '' || password === '' || PADDED_COLON.test(password)) return null
  return [head, login, password]
}

const SLASH_PASSWORD = /^(https?:\/\/[^/\s:@]+(?::\d{1,5})?):([^/\s:]+):(\S*\/\S*)$/i

/**
 * "https://shop.example.test:alice:Pass/word1": the first slash after the host belongs to the password, not to a path. The existing split takes the
 * first slash for the end of the url and finds one field too few. The password must not look like a URL fragment, and a number right after the host is
 * left to splitNumericLogin.
 */
export function splitSlashInPassword(line: string): [string, string, string] | null {
  const m = SLASH_PASSWORD.exec(line)
  if (!m) return null
  const [, url, login, password] = m
  if (/^\d{1,5}$/.test(login) && Number(login) <= 65535) return null
  if (looksLikeUrlFragment(password)) return null
  return [url, login, password]
}

const NUMERIC_LOGIN = /^(https?:\/\/[^/\s:@]+):(\d{1,5}):([^/\s:]+)$/i

/**
 * "https://shop.example.test:48213:Secret1": the existing split reads 48213 as a port (a number of up to five digits right after the host) and has
 * one field left, so the line is rejected. When the number is not a port a web server listens on, it is the login.
 */
export function splitNumericLogin(line: string): [string, string, string] | null {
  const m = NUMERIC_LOGIN.exec(line)
  if (!m) return null
  const n = Number(m[2])
  if (n > 65535 || COMMON_WEB_PORTS.has(n)) return null
  return [m[1], m[2], m[3]]
}

const PROTOCOL_RELATIVE = /^[^\s:@]+(?::\d{1,5}\/?[^\s:]*)?:[^\s:/]+:[^\s:]+$/

/**
 * "//shop.example.test/login:alice:Secret1": a protocol-relative url that parseLine used to take for a comment. Returns the line without its two slashes
 * (read as a scheme-less url, which is how such lines are stored), after a trailing ":http" / ":https" (a scheme some tools move to the end) is cut.
 * The host needs a dot, so "//TODO:fix:this" stays a comment.
 */
export function protocolRelativeToPlain(trimmed: string): string | null {
  if (!trimmed.startsWith('//') || trimmed.startsWith('///')) return null
  const rest = trimmed.slice(2).replace(/:https?$/i, '')
  if (!PROTOCOL_RELATIVE.test(rest)) return null
  const host = rest.split(/[/:]/)[0]
  return host.includes('.') ? rest : null
}

const ANDROID_WITHOUT_SCHEME = /^(\/\/)?[A-Za-z0-9_+-]{16,}==@[A-Za-z0-9_.-]+\/:/

/**
 * "HASH==@com.example.app/:alice@mail.test:Secret1": an Android app id that lost its "android://" prefix. Returns the line with the prefix put back, so it
 * is read like the lines that kept theirs.
 */
export function recoverAndroidAppLine(trimmed: string): string | null {
  const m = ANDROID_WITHOUT_SCHEME.exec(trimmed)
  if (!m) return null
  return m[1] ? `android:${trimmed}` : `android://${trimmed}`
}
```

```ts
function rng(seed: number): () => number {
  let a = seed >>> 0
  return () => {
    a = (a + 0x6d2b79f5) >>> 0
    let t = a

    t ^= t + Math.imul(t ^ (t >>> 7), t | 61)
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
  }
```

```ts
/** Email domain (lowercased): everything after the last '@'. Mirrors the SQL ED. */
export function emailDomainOf(email: string): string {
  const at = email.lastIndexOf('@')
  return at === -1 ? '' : email.slice(at + 1).toLowerCase()

/** Host of a URL: scheme/userinfo/port/path stripped, lowercased. Scans by index instead of splitting. */
function hostOf(url: string): string {
  let start = 0
  const scheme = url.indexOf('://')
  if (scheme !== -1) start = scheme + 3
  let end = url.length
  for (let i = start; i < url.length; i++) {
    const c = url.charCodeAt(i)
    if (c === 47 || c === 63 || c === 35) { end = i; break }       // / ? #
  }
  const at = url.lastIndexOf('@', end - 1)
  if (at >= start) start = at + 1                                   // drop userinfo
  for (let i = start; i < end; i++) {
    if (url.charCodeAt(i) === 58) { end = i; break }                // drop the port
  }
  return url.slice(start, end).toLowerCase()
}

/** URL top-level domain (last dotted label of the host). Mirrors lower(topLevelDomain(url)). */
export function urlTldOf(url: string): string {
  const h = hostOf(url)
  const dot = h.lastIndexOf('.')
  return dot === -1 ? '' : h.slice(dot + 1)
}

/**
 * Every suffix of the lists starts with '.', so `ed.endsWith(suffix)` holds exactly when the suffix equals one of the dot-suffixes of `ed`
 * (".example.co.uk", ".co.uk", ".uk"): a handful of Set lookups instead of a scan of the whole list.
 */
function matchesEmail(ed: string, suffixes: ReadonlySet<string>, providers: ReadonlySet<string>): boolean {
  if (providers.has(ed)) return true
  for (let i = ed.indexOf('.'); i !== -1; i = ed.indexOf('.', i + 1)) {
    if (suffixes.has(ed.slice(i))) return true
  }
  return false
}

/**
 * Classify a credential's country tier from its email + URL — identical logic and
 * precedence to buildCountryTierExpression() (email signal first: T1→T2→T3, then
 * URL-TLD fallback: T1→T2→T3, else untiered '').
 */
export function classifyTier(email: string, url: string): Tier {
  const ed = emailDomainOf(email)
  if (matchesEmail(ed, T1_SUFFIX_SET, T1_PROVIDER_SET)) return 'T1'
  if (matchesEmail(ed, T2_SUFFIX_SET, T2_PROVIDER_SET)) return 'T2'
  if (matchesEmail(ed, T3_SUFFIX_SET, T3_PROVIDER_SET)) return 'T3'
  const tld = urlTldOf(url)
  if (T1_TLD_SET.has(tld)) return 'T1'
  if (T2_TLD_SET.has(tld)) return 'T2'
  if (T3_TLD_SET.has(tld)) return 'T3'
  return ''
}

/**
 * The country (upper-case two-letter code) of the FIRST Tier 3 signal classifyTier uses: the e-mail provider or e-mail suffix, else the url TLD.
 * '' when classifyTier does not say T3. Used by the hard-keep policy (lib/ingest-filter.ts).
 */
export function t3CountryOf(email: string, url: string): string {
  const ed = emailDomainOf(email)
  if (matchesEmail(ed, T1_SUFFIX_SET, T1_PROVIDER_SET) || matchesEmail(ed, T2_SUFFIX_SET, T2_PROVIDER_SET)) return ''
  const provider = T3_PROVIDER_COUNTRY.get(ed)
  if (provider) return provider
  for (let i = ed.indexOf('.'); i !== -1; i = ed.indexOf('.', i + 1)) {
    const suffix = ed.slice(i)
    if (T3_SUFFIX_SET.has(suffix)) return suffix.slice(1).toUpperCase()
  }
  const tld = urlTldOf(url)
  if (T1_TLD_SET.has(tld) || T2_TLD_SET.has(tld)) return ''
  return T3_TLD_SET.has(tld) ? tld.toUpperCase() : ''
}
```

```ts
export const CSV_ROWS_PER_CHUNK = 2000

/** One CSV field for ClickHouse: wrapped in double quotes, inner quotes doubled. A value without a quote skips the regex. */
export function csvField(v: string): string {
  return v.indexOf('"') === -1 ? '"' + v + '"' : '"' + v.replace(/"/g, '""') + '"'
}

/** The six columns of ulp.credentials the importer writes, in the order insertBatch names them. `breachField` is csvField(breach_name), computed once. */
export function csvRow(c: ULPCredential, breachField: string): string {
  return csvField(c.url) + ',' + csvField(c.email) + ',' + csvField(c.password) + ',' + csvField(c.domain) + ',' + csvField(c.source_file) + ',' + breachField + '\n'
}

export function* csvChunks(credentials: ULPCredential[], breach_name: string, rowsPerChunk = CSV_ROWS_PER_CHUNK): Generator<string> {
  const breachField = csvField(breach_name)
  let chunk = ''
  let n = 0
  for (const c of credentials) {
    chunk += csvRow(c, breachField)
    if (++n === rowsPerChunk) { yield chunk; chunk = ''; n = 0 }
  }
  if (n > 0) yield chunk
}
```

```ts
/** Rows hashed per md5.update call: one update per field made ten native calls per row (1.61 us/row), one per 2000 rows is 0.44. */
const ROWS_PER_UPDATE = 2000

export function batchDedupToken(credentials: ULPCredential[], breach_name: string): string {
  const h = crypto.createHash('md5')
  let chunk = ''
  let n = 0
  for (const c of credentials) {
    // The same byte stream as one update per field: each field is followed by a separator, so no field can pair a surrogate with the next one.
    chunk += c.url + '\x00' + c.email + '\x00' + c.password + '\x00' + c.domain + '\x00' + c.source_file + '\x01'
    if (++n === ROWS_PER_UPDATE) { h.update(chunk); chunk = ''; n = 0 }
  }
  if (n > 0) h.update(chunk)
  h.update('\x02')
  h.update(breach_name)
  return h.digest('hex')
}
```

```ts
/**
 * The side copy: every line of an original file that the importer does NOT store, kept as raw bytes in a zstd file, so that the 140 GiB of originals can
 * go and a later change of policy or parser can still be applied to what was dropped (the T3 lines outside the keep-list, the rejects).
 *
 * A line is "stored" when parseInlineLine (the same call the stream parser makes for a line on its own) gives a credential. Lines that only the stateful
 * layers of the stream parser store (a label-free url / login / password record, a labeled block) read as dropped here, so the copy can hold a few lines
 * more than the importer rejected; it never holds fewer. Blank and white-space-only lines are not counted and not kept.
 */
import { createHash } from 'node:crypto'
import { createReadStream, createWriteStream } from 'node:fs'
import { once } from 'node:events'
import { pipeline } from 'node:stream/promises'
import { constants as zlibConstants, createZstdCompress } from 'node:zlib'
import { parseInlineLine } from '@/lib/ulp-parser'
import { repairUtf8Line } from '@/lib/ulp-utf8'

const NEWLINE = Buffer.from('\n')

export interface DropStats {
  lines: number
  stored: number
  dropped: number
  reasons: Record<string, number>
  outBytes: number
  sha256: string
}

export async function extractDroppedLines(
  srcPath: string,
  outPath: string,
  shouldHardDrop?: (email: string, url: string) => boolean,
): Promise<DropStats> {
  const stats: DropStats = { lines: 0, stored: 0, dropped: 0, reasons: {}, outBytes: 0, sha256: '' }
  const zstd = createZstdCompress({ params: { [zlibConstants.ZSTD_c_compressionLevel]: 3 } })
  const hash = createHash('sha256')
  zstd.on('data', (b: Buffer) => { hash.update(b); stats.outBytes += b.length })
  const written = pipeline(zstd, createWriteStream(outPath))

  let pending: Buffer[] = []
  let pendingBytes = 0
  const flush = async () => {
    if (pending.length === 0) return
    const buf = Buffer.concat(pending)
    pending = []; pendingBytes = 0
    if (!zstd.write(buf)) await once(zstd, 'drain')
  }
  const handle = (raw: Buffer) => {
    const text = repairUtf8Line(raw.toString('latin1'))
    if (!text.trim()) return
    stats.lines++
    const r = parseInlineLine(text, 'side-copy', shouldHardDrop)
    if (r.credential) { stats.stored++; return }
    stats.dropped++
    const reason = r.reason ?? 'unknown'
    stats.reasons[reason] = (stats.reasons[reason] ?? 0) + 1
    pending.push(raw, NEWLINE)
    pendingBytes += raw.length + 1
  }

  let carry: Buffer = Buffer.alloc(0)
  for await (const chunk of createReadStream(srcPath, { highWaterMark: 4 << 20 })) {
    let buf: Buffer = carry.length ? Buffer.concat([carry, chunk as Buffer]) : (chunk as Buffer)
    let start = 0
    for (let nl = buf.indexOf(10, start); nl !== -1; nl = buf.indexOf(10, start)) {
      handle(buf.subarray(start, nl))
      start = nl + 1
    }
    carry = Buffer.from(buf.subarray(start))
    if (pendingBytes >= 8 << 20) await flush()
  }
  if (carry.length) handle(carry)
  await flush()
  zstd.end()
  await written
  stats.sha256 = hash.digest('hex')
  return stats
}
```

```ts
/** Names the reason the result for one line differs between a baseline parser and parser v6 (scripts/parser-diff.ts). "unexplained" must stay at 0. */
import { protocolRelativeToPlain, recoverAndroidAppLine, splitNumericLogin, splitSlashInPassword, splitSpacePaddedColons } from '@/lib/ulp-recover'
import { repairUtf8Line } from '@/lib/ulp-utf8'

export interface Outcome { credential: { url: string; email: string; password: string } | null; reason: string | null }

export type ChangeClass =
  | 'same' | 'padded-colons' | 'android-no-scheme' | 'slash-in-password' | 'numeric-login' | 'protocol-relative'
  | 'utf8-recovered' | 'utf8-value' | 'login-equals-password' | 't3-kept' | 'unexplained'

export const CHANGE_CLASSES: ChangeClass[] = [
  'padded-colons', 'android-no-scheme', 'slash-in-password', 'numeric-login', 'protocol-relative',
  'utf8-recovered', 'utf8-value', 'login-equals-password', 't3-kept', 'unexplained',
]

type Cred = NonNullable<Outcome['credential']>
const sameCred = (a: Cred, b: Cred) => a.url === b.url && a.email === b.email && a.password === b.password

/** `oldOnRepaired` is the baseline parser run on the UTF-8-repaired line: when it already gives the new result, the repair alone explains the change. */
export function classifyChange(rawLine: string, oldO: Outcome, newO: Outcome, oldOnRepaired: Outcome): ChangeClass {
  if (oldO.credential && newO.credential) {
    if (sameCred(oldO.credential, newO.credential)) return 'same'
    return oldOnRepaired.credential && sameCred(oldOnRepaired.credential, newO.credential) ? 'utf8-value' : 'unexplained'
  }
  if (!oldO.credential && !newO.credential) return 'same'
  if (oldO.credential) return 'unexplained'                       // the old parser stored it and v6 does not
  const stored = newO.credential as Cred
  if (oldO.reason === 'tier_dropped') return 't3-kept'
  if (oldOnRepaired.credential && sameCred(oldOnRepaired.credential, stored)) return 'utf8-recovered'
  const t = repairUtf8Line(rawLine).trim()
  if (splitSpacePaddedColons(t)) return 'padded-colons'
  if (recoverAndroidAppLine(t)) return 'android-no-scheme'
  if (protocolRelativeToPlain(t)) return 'protocol-relative'
  if (splitSlashInPassword(t)) return 'slash-in-password'
  if (splitNumericLogin(t)) return 'numeric-login'
  if (stored.email === stored.password && !stored.email.includes('@')) return 'login-equals-password'
  return 'unexplained'
}
```

```bash
  # 1. a failure stops the feeding (in-flight files keep going; look at processing_jobs.error_message before retrying)
  nfail=$(( $(count inbox/failed) - fail0 ))
  if [ "$nfail" -gt 0 ]; then echo "$(now) STOP new file(s) in inbox/failed: $(ls -A inbox/failed | head -4 | tr '\n' ';')"; exit 2; fi

  top=$(ls -A inbox | grep -v -E '^(done|failed|processing)$' | wc -l); proc=$(count inbox/processing)
  inflight=$((top + proc))
  free=$(df -BG --output=avail / | tail -1 | tr -d ' G')
  pgib=$(ch "SELECT round(sumIf(bytes_on_disk, partition='$PARTITION')/1073741824, 1) FROM system.parts WHERE database='ulp' AND table='credentials' AND active"); pgib=${pgib:-0}

  # 2. limits
  phase_hit=0
  if [ "$PHASE_STOP_GIB" != 0 ] && awk -v a="$pgib" -v b="$PHASE_STOP_GIB" 'BEGIN{exit !(a>=b)}'; then phase_hit=1; fi
  if [ "$free" -lt "$FLOOR_GIB" ] && [ "$paused_disk" = 0 ]; then paused_disk=1; echo "$(now) LOW DISK free=${free}GiB < ${FLOOR_GIB}: feeding paused"; fi
  if [ "$paused_disk" = 1 ] && [ "$free" -ge $((FLOOR_GIB + 5)) ]; then paused_disk=0; echo "$(now) DISK OK free=${free}GiB: feeding resumes"; fi

  # 3. feed
  while [ "$inflight" -lt "$IN_FLIGHT" ] && [ "$fed" -lt "$total" ] && [ "$paused_disk" = 0 ] && [ "$phase_hit" = 0 ]; do
    idx=$((idx + 1)); line=$(sed -n "${idx}p" "$WORK/queue.tsv"); [ -z "$line" ] && break
    size=${line%%$'\t'*}; name=${line#*$'\t'}
    if ln "$PILE_DIR/$name" "inbox/$name" 2>/dev/null; then
      fed=$((fed + 1)); inflight=$((inflight + 1))
      echo "$(now) FEED $fed/$total $(awk -v s="$size" 'BEGIN{printf "%.1f MB", s/1048576}') $name"
    else
      echo "$(now) STOP could not link: $name"; exit 3
    fi
  done

  # 4. progress / jobs summary every 5 minutes
  t=$(date +%s)
  if [ $((t - last_progress)) -ge 300 ]; then
    last_progress=$t
    echo "$(now) PROGRESS fed=$fed/$total inflight=$inflight done=+$(( $(count inbox/done) - done0 )) free=${free}GiB p${PARTITION}=${pgib}GiB"
  fi
  if [ $((t - last_jobs)) -ge 300 ]; then
    last_jobs=$t
    rm -rf "$WORK/dbcopy"; mkdir -p "$WORK/dbcopy"; cp data/ulp.db data/ulp.db-wal data/ulp.db-shm "$WORK/dbcopy/" 2>/dev/null
    node scripts/pile/jobs-summary.cjs "$WORK/dbcopy" "$START_TS" 2>&1 | head -1 | sed "s/^/$(now) /"
  fi

  # 5. end conditions
  if [ "$inflight" = 0 ] && { [ "$fed" -ge "$total" ] || [ "$phase_hit" = 1 ]; }; then
    if [ "$phase_hit" = 1 ]; then echo "$(now) PHASE LIMIT reached p${PARTITION}=${pgib}GiB, nothing in flight: feeding stopped after $fed files"; else echo "$(now) COMPLETE all $fed files processed"; fi
    rm -rf "$WORK/dbcopy"; mkdir -p "$WORK/dbcopy"; cp data/ulp.db data/ulp.db-wal data/ulp.db-shm "$WORK/dbcopy/" 2>/dev/null
    node scripts/pile/jobs-summary.cjs "$WORK/dbcopy" "$START_TS" 2>&1 | head -1 | sed "s/^/$(now) /"
    exit 0
  fi
  sleep 10
done
```

```js
// usage: node scripts/pile/jobs-summary.cjs <dir holding a COPY of ulp.db (+ -wal, -shm)> <since 'YYYY-MM-DD HH:MM:SS' UTC> [list]
// One summary line of the inbox import jobs recorded since <since>; with "list" also one line per job (never a line of a file).
const Database = require('better-sqlite3')
const [dir, since, mode] = process.argv.slice(2)
if (!dir || !since) { console.error('usage: jobs-summary.cjs <dir> <since UTC> [list]'); process.exit(2) }
const db = new Database(`${dir}/ulp.db`)
const rows = db.prepare('SELECT source, filename, status, imported, known, skipped, duration_ms, error_message, created_at FROM processing_jobs WHERE created_at >= ? ORDER BY created_at').all(since)
const sum = (k, st) => rows.filter(r => !st || r.status === st).reduce((a, r) => a + (Number(r[k]) || 0), 0)
const n = st => rows.filter(r => r.status === st).length
console.log(
  `JOBS since ${since}: done=${n('done')} failed=${n('failed')} imported(new)=${sum('imported', 'done')} known=${sum('known', 'done')} skipped=${sum('skipped', 'done')} ` +
  `busy_min=${(sum('duration_ms', 'done') / 60000).toFixed(1)}`,
)
if (mode === 'list') {
  for (const r of rows) console.log(`  ${r.created_at} ${r.status.padEnd(6)} new=${r.imported} known=${r.known} skipped=${r.skipped} ${(r.duration_ms / 1000).toFixed(1)}s ${r.filename}${r.error_message ? '  ERR: ' + String(r.error_message).slice(0, 160) : ''}`)
}
```

```bash
    n_ok=$((n_ok+1)); printf '%s\n' "$f" >> "$WORK/reset-unlink.txt"

echo "== after"
echo "partition $PARTITION rows: $(ch "SELECT count() FROM ulp.credentials WHERE _partition_id='$PARTITION'")  sources since the cut: $(ch "SELECT count() FROM ulp.sources WHERE imported_at >= '$CUT'")"
```

```bash
#!/bin/bash
# One monitor for a pile run: feeder events (STOP / LOW DISK / DISK OK / PHASE LIMIT / COMPLETE), anomalies (app or ClickHouse restart or unhealthy, read-only
# replica, replication queue, open mutations, detached parts, errors in text_log, app log errors) and a HEARTBEAT line at the start and every HB seconds.
# usage: START_TS='2026-10-12 12:00:00' FEED_LOG=<path of the feeder's output> scripts/pile/watch.sh      (run it under the Monitor tool; it ends only when stopped)
# Env: START_TS (required), FEED_LOG (required), HB (1800), PARTITION (202610), CH_CONTAINER, APP_CONTAINER (ulpsuite_app), PILE_WORK_DIR.
REPO=$(cd "$(dirname "$0")/../.." && pwd); cd "$REPO" || exit 1
WORK=${PILE_WORK_DIR:-"$REPO/.pile-work"}; mkdir -p "$WORK"
CH_CONTAINER=${CH_CONTAINER:-ulpsuite_clickhouse}; APP_CONTAINER=${APP_CONTAINER:-ulpsuite_app}
PARTITION=${PARTITION:-202610}
START_TS=${START_TS:?set START_TS}; FEED_LOG=${FEED_LOG:?set FEED_LOG}
HB=${HB:-1800}
trap 'kill $(jobs -p) 2>/dev/null' EXIT

tail -n 0 -F "$FEED_LOG" | grep --line-buffered -E 'STOP|LOW DISK|DISK OK|PHASE LIMIT|COMPLETE' | sed -u 's/^/FEEDER /' &
docker logs -f --since "$(date -u +%Y-%m-%dT%H:%M:%SZ)" "$APP_CONTAINER" 2>&1 \
  | grep -E --line-buffered 'rror|[Ff]ailed|CRITICAL|[Pp]aused|abort|ECONN' | grep -v --line-buffered 'monitor-rescan' | sed -u 's/^/APP-LOG /' &

app0=$(docker inspect -f '{{.State.StartedAt}}' "$APP_CONTAINER"); ch0=$(docker inspect -f '{{.State.StartedAt}}' "$CH_CONTAINER")
chq() { docker exec "$CH_CONTAINER" clickhouse-client -q "$1" 2>/dev/null; }
heartbeat() {
  local fed js p free inflight alive
  fed=$(grep -c ' FEED ' "$FEED_LOG" 2>/dev/null)
  rm -rf "$WORK/hbq"; mkdir -p "$WORK/hbq"; cp data/ulp.db data/ulp.db-wal data/ulp.db-shm "$WORK/hbq/" 2>/dev/null
  js=$(node scripts/pile/jobs-summary.cjs "$WORK/hbq" "$START_TS" 2>&1 | head -1 | sed 's/^JOBS since [0-9: -]*: //')
  p=$(chq "SELECT concat(toString(round(sumIf(bytes_on_disk, partition='$PARTITION')/1073741824, 1)), ' GiB, ', toString(sumIf(rows, partition='$PARTITION')), ' rows') FROM system.parts WHERE database='ulp' AND table='credentials' AND active")
  free=$(df -BG --output=avail / | tail -1 | tr -d ' ')
  inflight=$(( $(ls -A inbox | grep -v -E '^(done|failed|processing)$' | wc -l) + $(ls -A inbox/processing | wc -l) ))
  pgrep -f "scripts/pile/feed.sh" >/dev/null && alive=alive || alive=NOT-RUNNING
  echo "$(date -u +%T) HEARTBEAT feeder=$alive fed=$fed inflight=$inflight | $js | p${PARTITION}=$p | free=$free"
}

prev=""; last_hb=0
while true; do
  now=$(date +%s)
  if [ $((now - last_hb)) -ge "$HB" ]; then heartbeat; last_hb=$now; fi
  a=$(docker inspect -f '{{.State.StartedAt}}|{{.State.Health.Status}}' "$APP_CONTAINER" 2>&1)
  c=$(docker inspect -f '{{.State.StartedAt}}|{{.State.Health.Status}}' "$CH_CONTAINER" 2>&1)
  q=$(chq "SELECT concat((SELECT toString(max(is_readonly)) FROM system.replicas), '|', (SELECT toString(max(queue_size)) FROM system.replicas), '|', toString((SELECT count() FROM system.mutations WHERE NOT is_done)), '|', toString((SELECT count() FROM system.detached_parts)), '|', toString((SELECT count() FROM system.text_log WHERE level IN ('Error','Fatal') AND event_time > now() - INTERVAL 10 MINUTE)))")
  IFS='|' read -r ro qs mut det err <<< "$q"
  issues=""
  [ "${a%%|*}" != "$app0" ] && issues="$issues app-restarted"
  [ "${a##*|}" != "healthy" ] && issues="$issues app-${a##*|}"
  [ "${c%%|*}" != "$ch0" ] && issues="$issues clickhouse-restarted"
  [ "${c##*|}" != "healthy" ] && issues="$issues clickhouse-${c##*|}"
  [ "$ro" != "0" ] && issues="$issues replica-readonly=$ro"
  [ "${qs:-0}" -gt 20 ] 2>/dev/null && issues="$issues repl-queue=$qs"
  [ "${mut:-0}" != "0" ] && issues="$issues open-mutations=$mut"
  [ "${det:-0}" != "0" ] && issues="$issues detached-parts=$det"
  [ "${err:-0}" != "0" ] && issues="$issues text_log-errors-10m=$err"
  pgrep -f "scripts/pile/feed.sh" >/dev/null || issues="$issues feeder-not-running"
  if [ "$issues" != "$prev" ]; then
    if [ -n "$issues" ]; then echo "$(date -u +%T) ANOMALY:$issues"; else echo "$(date -u +%T) ALL CLEAR (was:$prev)"; fi
    prev="$issues"
  fi
  sleep 60
done
```
